Interactive Code Refactoring Showcase

Inspect how Edge Clean Copilot decouples a coupled Express controller into Domain, Application, and Port contracts.

✓ Clean Architecture Invariants Applied: Zero ORM in domain • Invariant guarded in constructor • Abstract IOrderRepository port • Functional Result monad. 
// 1. DOMAIN LAYER: Pure Business Entity (src/domain/entities/Order.ts)
export class Order {
  private constructor(
    public readonly id: string,
    public readonly customerId: string,
    public readonly total: number,
    public readonly status: 'PENDING' | 'PAID'
  ) {}

  public static create(customerId: string, items: OrderItem[]): Order {
    if (!items || items.length === 0) {
      throw new DomainError('Order must contain at least one item');
    }
    const total = items.reduce((sum, item) => sum + item.price * item.quantity, 0);
    return new Order(crypto.randomUUID(), customerId, total, 'PENDING');
  }
}

// 2. APPLICATION LAYER: Port Contract (src/application/ports/IOrderRepository.ts)
export interface IOrderRepository {
  save(order: Order): Promise<void>;
}

// 3. APPLICATION LAYER: Use Case (src/application/use-cases/CreateOrderUseCase.ts)
export class CreateOrderUseCase {
  constructor(private readonly orderRepo: IOrderRepository) {}

  async execute(dto: CreateOrderDTO): Promise<Result<Order, DomainError>> {
    try {
      const order = Order.create(dto.customerId, dto.items);
      await this.orderRepo.save(order);
      return Result.ok(order);
    } catch (err) {
      return Result.fail(err as DomainError);
    }
  }
}
✕ Architectural Anti-Patterns Detected: Direct SQL query inside HTTP route • Zero defensive constructor • Express HTTP tightly coupled to business rules. 
// Coupled Legacy Code (Direct Express + SQL in Handler)
import express from 'express';
import { db } from './database';

app.post('/api/orders', async (req: any, res: any) => {
  const { customerId, items } = req.body;
  
  // Inline validation mixed with HTTP presentation logic
  if (!items || items.length === 0) {
    return res.status(400).json({ error: 'Items required' });
  }

  const total = items.reduce((sum: any, i: any) => sum + i.price * i.qty, 0);

  // Direct database leak in web controller
  const result = await db.query(
    'INSERT INTO orders (customer_id, total, status) VALUES ($1, $2, $3) RETURNING id',
    [customerId, total, 'PENDING']
  );

  res.status(201).json({ orderId: result.rows[0].id, total });
});
